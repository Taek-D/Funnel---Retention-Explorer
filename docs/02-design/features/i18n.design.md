### I18N-1: i18next Infrastructure Setup

#### 2.1.1 Dependencies (package.json)

**Add to dependencies:**

```json

  "i18next": "^24.0.0",

}
```

#### 2.1.2 i18n Config (`lib/i18n.ts`)

```typescript
import i18n from 'i18next';
import { initReactI18next } from 'react-i18next';
import LanguageDetector from 'i18next-browser-languagedetector';

import koCommon from '../locales/ko/common.json';
import koPages from '../locales/ko/pages.json';
import koInsights from '../locales/ko/insights.json';
import enCommon from '../locales/en/common.json';
import enPages from '../locales/en/pages.json';
import enInsights from '../locales/en/insights.json';

i18n
  .use(LanguageDetector)
  .use(initReactI18next)
  .init({
    resources: {
      ko: { common: koCommon, pages: koPages, insights: koInsights },
      en: { common: enCommon, pages: enPages, insights: enInsights },
    },
    fallbackLng: 'ko',
    defaultNS: 'common',
    ns: ['common', 'pages', 'insights'],
    interpolation: { escapeValue: false },
    detection: {
      order: ['localStorage', 'navigator'],
      lookupLocalStorage: 'fre-language',
      caches: ['localStorage'],
    },
    react: { useSuspense: false },
  });

export default i18n;
```

**Design Decisions:**
- `fallbackLng: 'ko'` — 누락 키는 한국어로 표시 (기존 동작 유지)
- `useSuspense: false` — 비동기 로딩 없이 번들에 포함 (초기 로드 지연 없음)
- 번역 JSON을 static import — 별도 HTTP 요청 없음, tree-shaking 가능
- `detection.order: ['localStorage', 'navigator']` — 사용자 선택 > 브라우저 언어

#### 2.1.3 Entry Point (`index.tsx`)

```typescript
// Add as first import (before React)
import './lib/i18n';
```

// After:
import i18n from './i18n';
const t = i18n.t.bind(i18n);

insights.push({
  title: t('insights:maxDropoff.title'),
  body: t('insights:maxDropoff.body', { step, rate, count }),
  recommendations: [t('insights:maxDropoff.rec1')]

### I18N-5: Language Switcher UI

**File**: `components/LanguageSwitcher.tsx` (NEW)

#### 2.5.1 Component Design

```tsx
import React from 'react';
import { useTranslation } from 'react-i18next';
import { Globe } from './Icons';

export const LanguageSwitcher: React.FC = () => {
  const { i18n } = useTranslation();

  const toggle = () => {
    const next = i18n.language === 'ko' ? 'en' : 'ko';
    i18n.changeLanguage(next);
    document.documentElement.lang = next;
  };

  const label = i18n.language === 'ko' ? 'EN' : 'KO';

  return (
    <button
      onClick={toggle}
      className="w-10 h-10 flex items-center justify-center rounded-md text-slate-500 hover:text-white hover:bg-white/[0.03] transition-colors text-[11px] font-mono font-semibold"
      aria-label={`Switch to ${i18n.language === 'ko' ? 'English' : '한국어'}`}
      title={i18n.language === 'ko' ? 'English' : '한국어'}
    >
      {label}
    </button>
  );
};
```
